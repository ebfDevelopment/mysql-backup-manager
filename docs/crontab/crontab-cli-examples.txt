# ========================================
# Exemplos de Crontab para Backups MySQL
# ========================================

# Como editar o crontab:
# crontab -e

# Formato:
# * * * * * comando
# │ │ │ │ │
# │ │ │ │ └─── Dia da semana (0-7, 0 e 7 = Domingo)
# │ │ │ └───── Mês (1-12)
# │ │ └─────── Dia do mês (1-31)
# │ └───────── Hora (0-23)
# └─────────── Minuto (0-59)

# ========================================
# 1. BACKUP DIÁRIO
# Todo dia às 2h da manhã
# ========================================
0 2 * * * /usr/bin/php /var/www/html/cron/daily-backup.php >> /var/log/backup-daily.log 2>&1

# ========================================
# 2. BACKUP SEMANAL
# Todo domingo às 3h
# ========================================
0 3 * * 0 /usr/bin/php /var/www/html/cron/weekly-backup.php >> /var/log/backup-weekly.log 2>&1

# ========================================
# 3. BACKUP MÚLTIPLOS BANCOS
# Todo dia à 1h
# ========================================
0 1 * * * /usr/bin/php /var/www/html/cron/multiple-databases-backup.php >> /var/log/backup-multiple.log 2>&1

# ========================================
# 4. LIMPEZA DE BACKUPS ANTIGOS
# Todo dia às 4h
# ========================================
0 4 * * * /usr/bin/php /var/www/html/cron/cleanup-old-backups.php >> /var/log/backup-cleanup.log 2>&1

# ========================================
# 5. BACKUP A CADA 6 HORAS
# Às 0h, 6h, 12h e 18h
# ========================================
0 */6 * * * /usr/bin/php /var/www/html/cron/daily-backup.php >> /var/log/backup-6h.log 2>&1

# ========================================
# 6. BACKUP A CADA HORA (Horário Comercial)
# De segunda a sexta, das 8h às 18h
# ========================================
0 8-18 * * 1-5 /usr/bin/php /var/www/html/cron/daily-backup.php >> /var/log/backup-hourly.log 2>&1

# ========================================
# 7. BACKUP MENSAL
# Primeiro dia de cada mês às 1h
# ========================================
0 1 1 * * /usr/bin/php /var/www/html/cron/weekly-backup.php >> /var/log/backup-monthly.log 2>&1

# ========================================
# DICAS
# ========================================

# Verificar logs do cron:
# tail -f /var/log/cron
# tail -f /var/log/backup-daily.log

# Testar manualmente antes de agendar:
# /usr/bin/php /var/www/html/cron/daily-backup.php

# Ver tarefas agendadas:
# crontab -l

# Remover todas as tarefas:
# crontab -r

# Editar para outro usuário (como root):
# crontab -u www-data -e
